# Mandarin Learning Reader — Python dependencies
# Python 3.9+.  Install with:  pip install -r requirements.txt

# ── Core (app is useless without these) ──────────────────────────────
edge-tts                      # Mandarin TTS (free MS Edge voices, online)
ebooklib                      # read EPUB spine / TOC / metadata
beautifulsoup4                # HTML -> clean text extraction
pypinyin                      # per-character tone-marked pinyin
faster-whisper                # offline local speech-to-text (recommended)
sounddevice                   # microphone capture
scipy                         # write WAV for Whisper
pygame                        # play sentence audio (GUI)

# ── Optional ─────────────────────────────────────────────────────────
opencc-python-reimplemented   # Simplified <-> Traditional toggle + grade norm
